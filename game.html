import random
def play():
options = ["a", "b", "c"]
print("Enter rock(a), paper(b) or scissors(c) : ")
user_choice = input("User : ")
if user_choice not in options:
print("Try rock(a), paper(b) or scissors(c) . ")
return
pc_choice = random.choice(options)
print(f"Computer : {pc_choice}")
if (user_choice == "b" and pc_choice == "a")or \
(user_choice == "a" and pc_choice == "c") or \
(user_choice == "c" and pc_choice == "b") :
print("You win 😁 ! ")
elif user_choice == pc_choice:
print("Tie 😊 ! ")
else:
print("Computer wins 😎 ")

play()
